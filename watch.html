function getSubtitleUrl() { if (!currentDramaData) return null; const drama = subtitlesData.dramas.find( item => String(item.id).toLowerCase() === String(dramaId).toLowerCase() ); if (!drama) return null; const season = drama.seasons?.[currentSeasonKey]; if (!season) return null; const originalUrl = season.episodes?.[String(currentEpNum)]; if (!originalUrl) return null; try { const url = new URL(originalUrl); const parts = url.pathname.split("/"); const downloadIndex = parts.indexOf("download"); if (downloadIndex !== -1) { const releaseTag = parts[downloadIndex + 1]; const fileName = parts.slice(downloadIndex + 2).join("/"); return ( "https://raw.githubusercontent.com/moanahabhahaha-debug/new4/main/subtitles/" + encodeURIComponent(releaseTag) + "/" + fileName ); } } catch (error) { console.error("Subtitle URL Error:", error); } return null; }
